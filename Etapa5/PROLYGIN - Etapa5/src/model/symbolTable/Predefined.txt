class Integer {
    int value;
    Integer(int v) {
        value = v;
    }
    int intValue() { return value; }
}

class Boolean {
    boolean value;
    Boolean(boolean v) {
        value = v;
    }
    boolean booleanValue() { return value; }
}

class Character {
    char value;
    Character(char v) {
        value = v;
    }
    char charValue() { return value; }
}
